        always@(posedge CLK)
            begin
                    r_nwrt <= NWRT;
                    r_nce <= NCE;
                    r_din <= DIN;
                    r_addr <= A;
                    
            #0.1
                    if(!r_nce && !r_nwrt)
                    -> write;
                    if(!r_nce && r_nwrt)
                    -> read;
                    
                end
